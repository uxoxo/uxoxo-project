\newcommand{\uxpart}[1]{%
  \clearpage
  \par\addvspace{0.3\baselineskip}%
  \addcontentsline{toc}{section}{#1}%
  \noindent{\large\bfseries\color{cSig!55!black}#1}\par
  \nobreak\vspace{1pt}{\color{cSig!35}\hrule height 0.9pt}%
  \addvspace{0.75\baselineskip}\nobreak
  \uxnocleartrue}

\newcommand{\sectionnotation}[1]{%
  \par\nobreak\smallskip
  \setlength{\fboxsep}{5pt}%
  \noindent\colorbox{cInk!4}{%
    \begin{minipage}{\dimexpr\linewidth-2\fboxsep\relax}
      \small\textbf{\textsf{\textcolor{cInk!75}{New in this section.}}}\enspace #1
    \end{minipage}}%
  \par\smallskip}
